**Create apps/tui/src/help.rs:**

Create a new module that defines key binding data and renders the help overlay.

1. Define key binding data as structured constants (not just strings) so content is testable:

```rust
pub struct KeyBinding {
    pub keys: &'static str,    // e.g. "[j/k]" or "[Tab]"
    pub description: &'static str,
}

pub struct KeySection {
    pub title: &'static str,
    pub bindings: &'static [KeyBinding],
}
```

2. Define key sections:
- **Global** (always active): `[q]` Quit, `[?]` Help, `[Tab]` Next pane, `[Shift+Tab]` Previous pane, `[Esc]` Back to tree
- **Tree Pane**: `[j/k]` or `[Up/Down]` Navigate, `[Enter]` Expand/start session, `[r]` Start session, `[R]` Restart session, `[x]` Stop session
- **Output Pane**: `[j/k]` or `[Up/Down]` Scroll line, `[Page Up/Down]` Scroll page, `[g/Home]` Top, `[G/End]` Bottom, `[z]` Zoom toggle, `[i]` Compose
- **Composer**: `[Enter]` Send message, `[Shift+Enter]` New line, `[Ctrl+C]` Clear input

3. Render function `pub fn render_help_overlay(frame: &mut Frame, focus: Focus)`:
- Use `centered_rect(60, 70, frame.area())` helper to position overlay at 60% width, 70% height
- Render `Clear` widget first to blank the area
- Dark background box (`Block` with `Borders::ALL`, cyan border, title " Help ")
- Show current pane at top: "Current: Tree Pane" (or Output/Composer) with cyan highlight
- For each section: render section title as a bold header line, then key bindings below
- Highlight the section matching current focus with cyan text, other sections in normal white
- Key format: `[Tab] Switch pane` with the bracketed key in cyan, description in white

```rust
fn centered_rect(percent_x: u16, percent_y: u16, area: Rect) -> Rect {
    let popup_layout = Layout::vertical([
        Constraint::Percentage((100 - percent_y) / 2),
        Constraint::Percentage(percent_y),
        Constraint::Percentage((100 - percent_y) / 2),
    ]).split(area);
    Layout::horizontal([
        Constraint::Percentage((100 - percent_x) / 2),
        Constraint::Percentage(percent_x),
        Constraint::Percentage((100 - percent_x) / 2),
    ]).split(popup_layout[1])[1]
}
```

**render_zoomed function:**

```rust
fn render_zoomed(frame: &mut ratatui::Frame, app: &mut App) {
    let composer_height = if app.selected_workspace()
        .and_then(|ws| app.state.find_session_by_workspace(&ws.id))
        .map(|s| s.status == SessionStatus::Running)
        .unwrap_or(false)
    {
        app.composer.height_hint()
    } else {
        3 // minimal height for "Press R to resume" hint
    };

    let chunks = Layout::vertical([
        Constraint::Min(1),              // output (fills remaining)
        Constraint::Length(composer_height), // composer
        Constraint::Length(1),           // status bar
    ]).split(frame.area());

    // Render output (reuse the output rendering from render_right_pane but full-width)
    // Extract the selected workspace info
    let ws_info = app.selected_workspace().cloned().and_then(|ws| {
        app.state.find_session_by_workspace(&ws.id)
            .map(|s| (ws, s.id.clone(), s.status.clone()))
    });

    if let Some((ws, _session_id, session_status)) = ws_info {
        // Output area - same rendering as render_right_pane but in chunks[0]
        let output_area = chunks[0];
        let inner_height = output_area.height.saturating_sub(2) as usize;
        let inner_width = output_area.width.saturating_sub(3) as usize; // borders + scrollbar

        let scrollback = app.scrollbacks.get(&ws.id);
        let visible = scrollback.map(|buf| buf.visible_lines(inner_height)).unwrap_or_default();

        // Chat bubble styling (same as Plan 02 -- reuse the render_output_content helper)
        // DRY: Extract a render_output_content(frame, area, visible, inner_width, focus, ws_name, session_status) helper
        // that both render_right_pane and render_zoomed call for the output paragraph + scrollbar.

        let output_border_style = if app.focus == Focus::Output {
            Style::default().fg(Color::Cyan)
        } else {
            Style::default().fg(Color::DarkGray)
        };
        let output_block = Block::default()
            .title(format!(" {} ", ws.name))
            .borders(Borders::ALL)
            .border_style(output_border_style);
        // Render paragraph with chat bubble lines + block + wrap
        // Render scrollbar

        // Composer area
        let composer_area = chunks[1];
        if session_status == SessionStatus::Running {
            frame.render_widget(app.composer.widget(), composer_area);
            // Status text overlay (same as Plan 02)
        } else {
            let hint = Paragraph::new("Press 'R' to resume session")
                .style(Style::default().fg(Color::DarkGray))
                .block(Block::default().title(" Composer ").borders(Borders::ALL).border_style(Style::default().fg(Color::DarkGray)));
            frame.render_widget(hint, composer_area);
        }

        // Status bar (bottom line)
        let status_bar_area = chunks[2];
        let (status_icon, status_color) = match session_status {
            SessionStatus::Running => ("\u{25B6}", Color::Green),
            SessionStatus::Stopped => ("\u{25A0}", Color::Yellow),
            SessionStatus::Failed => ("\u{2717}", Color::Red),
            SessionStatus::Exited => ("\u{2717}", Color::DarkGray),
        };

        let scroll_info = if let Some(buf) = app.scrollbacks.get(&ws.id) {
            let total = buf.total_lines();
            let offset = buf.clamped_offset(inner_height);
            let current_line = total.saturating_sub(offset);
            format!("line {}/{}", current_line, total)
        } else {
            String::new()
        };

        let status_line = Line::from(vec![
            Span::styled(format!(" {} ", ws.name), Style::default().fg(Color::White).add_modifier(Modifier::BOLD)),
            Span::styled(format!(" {} ", status_icon), Style::default().fg(status_color)),
            Span::raw(" "),
            Span::styled(scroll_info, Style::default().fg(Color::DarkGray)),
            Span::raw("  "),
            Span::styled("[z] exit zoom", Style::default().fg(Color::DarkGray)),
        ]);
        frame.render_widget(
            Paragraph::new(status_line).style(Style::default().bg(Color::DarkGray).fg(Color::White)),
            status_bar_area,
        );
    }

    app.last_output_height = chunks[0].height;
}
```
